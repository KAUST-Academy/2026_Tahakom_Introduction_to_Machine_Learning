\section{Learning Outcomes}
\begin{frame}{Learning Outcomes}
    \large
    By the end of this session, you will be able to:
    \vspace{0.5em}
    \begin{itemize}
        \setlength{\itemsep}{1em}
        \item Model a sequential decision problem as a \textbf{Markov decision process}: states, actions, rewards, transitions and a discount factor.
        \item Explain \textbf{value functions} and the \textbf{Bellman equation}, and train a tabular \textbf{Q-learning} agent with $\varepsilon$-greedy exploration.
        \item Identify \textbf{practical applications and limitations} of RL.
    \end{itemize}
\end{frame}
